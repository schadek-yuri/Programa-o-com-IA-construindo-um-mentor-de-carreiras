2. script.js O código abaixo funciona como uma demonstração local: ele compara os prompts, conta caracteres e palavras e explica por que isso não equivale à medição real de tokens. Para obter o consumo real, é necessário conectar o mentor da unidade ou uma API de IA. javascript const prompts = { curto: "Me ajude a escolher uma carreira em tecnologia.", detalhado: `Atue como mentor de carreiras em tecnologia. Faça uma análise das minhas habilidades, interesses e objetivos profissionais. Compare desenvolvimento web, ciência de dados e segurança da informação. Para cada área, apresente competências necessárias, ferramentas para estudar, um plano inicial de 30 dias e três projetos para montar um portfólio. Finalize com recomendações práticas para quem está começando.` }; const resultados = { curto: null, detalhado: null }; function analisarPrompt(texto) { return { caracteres: texto.length, palavras: texto.trim().split(/\s+/).length }; } function exibirResultado(tipo) { const prompt = prompts[tipo]; const analise = analisarPrompt(prompt); const destino = document.getElementById( tipo === "curto" ? "resultadoCurto" : "resultadoDetalhado" ); resultados[tipo] = analise; destino.replaceChildren(); const linhas = [ ["Caracteres", analise.caracteres], ["Palavras", analise.palavras], ["Tokens de entrada", "Não medidos"], ["Tokens de saída", "Não medidos"], ["Total de tokens", "Não medido"] ]; linhas.forEach(([rotulo, valor]) => { const linha = document.createElement("p"); const nome = document.createElement("strong"); nome.textContent = `${rotulo}: `; linha.append(nome, document.createTextNode(String(valor))); destino.appendChild(linha); }); atualizarComparacao(); } function atualizarComparacao() { const destino = document.getElementById("comparacao"); destino.replaceChildren(); if (!resultados.curto || !resultados.detalhado) { destino.textContent = "Execute os dois testes para visualizar a comparação."; return; } const curto = resultados.curto; const detalhado = resultados.detalhado; const diferencaCaracteres = detalhado.caracteres - curto.caracteres; const diferencaPalavras = detalhado.palavras - curto.palavras; const itens = [ `Diferença de caracteres: ${diferencaCaracteres}`, `Diferença de palavras: ${diferencaPalavras}`, "O prompt detalhado contém mais texto e tende a consumir mais tokens de entrada." ]; itens.forEach(texto => { const p = document.createElement("p"); p.textContent = texto; destino.appendChild(p); }); const aviso = document.createElement("p"); aviso.className = "aviso"; aviso.textContent = "A quantidade real de tokens de saída depende da resposta gerada pela IA. " + "Caracteres e palavras não substituem a contagem de tokens."; destino.appendChild(aviso); } document.getElementById("testeCurto").addEventListener("click", () => { exibirResultado("curto"); }); document.getElementById("testeDetalhado").addEventListener("click", () => { exibirResultado("detalhado"); });
